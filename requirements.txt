# CLIPOS dependencies.
#
# Use scripts/install_deps.py instead of installing this file directly:
#
#     python scripts/install_deps.py --build
#
# It installs the same packages and then repairs the OpenCV situation, which
# cannot be expressed here: scenedetect depends on "opencv-python", the GUI
# build, which pip treats as a different distribution from
# "opencv-python-headless" even though they overwrite each other's files.
#
# ---- Core UI ----
PySide6>=6.6,<6.9
psutil>=6,<8

# ---- Video processing ----
# FFmpeg/FFprobe binaries are vendored by scripts/vendor_ffmpeg.py (not pip)
av>=14,<19  # bounded NUT demux/decode for shared-graph composited playback

# ---- AI / inference ----
numpy>=1.24
onnxruntime>=1.16,<2
supervision==0.27.0
matplotlib>=3.7,<4
scipy>=1.10,<2
# OpenCV 5.x removed cv2.CascadeClassifier which CLIPOS uses for face detection.
#
# scenedetect depends on "opencv-python", the *GUI* build, which needs libGL
# and would replace the headless build below.  On a headless build machine
# (CI, a container, a server) importing it fails outright with
# "libGL.so.1: cannot open shared object file", so the headless build has to
# be installed after every other package, and the GUI build has to go.
#
# The reinstall below is what actually pins the version; the constraint alone
# is not enough because pip resolves "opencv-python" and
# "opencv-python-headless" as two unrelated distributions that share files.
opencv-python-headless>=4.8,<5
faster-whisper>=1.0,<2
scenedetect>=0.6.3
huggingface-hub>=0.23

# NOTE: PyTorch is deliberately NOT a dependency.
#
# Speech-to-text runs on faster-whisper, which uses CTranslate2 — a separate
# inference engine that needs no PyTorch at all.  PyTorch is only consulted
# to *detect* an NVIDIA or Apple GPU so the app can report it and offer a
# CUDA/MPS compute type; every use of it is wrapped in a try/except, so the
# app works exactly the same without it.
#
# Installing PyTorch pulls in several gigabytes of wheels, which makes the
# install slow and the release builds expensive.  Install it yourself if you
# want the GPU badge in Settings > AI Models:
#
#     pip install torch --index-url https://download.pytorch.org/whl/cpu
#
# The CPU path (CTranslate2 int8) is fast enough for real-time transcription
# on every supported machine, so this is a convenience, not a requirement.

# ---- Optional (local LLM via Ollama) ----
requests>=2.31

# ---- Packaging ----
pyinstaller>=6.6
pillow>=10.0

# ---- Windows installer ----
# ziglang carries an LLVM toolchain with a Windows target, so a native
# .exe installer can be cross-compiled from any host.  Optional: Inno Setup
# and NSIS are used instead when they are installed.
ziglang>=0.11; platform_system != "Windows"

# Public video imports (user-authorised content only)
yt-dlp>=2025.1.15
